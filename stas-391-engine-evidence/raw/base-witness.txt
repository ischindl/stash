      63	/**
      64	 * Reads the payload's text, or throws when the payload has no content at all.
      65	 *
      66	 * `[]` counts as absent: an array with no blocks states nothing, while
      67	 * `[{ type: "text", text: "" }]` and `""` state that zero bytes were delivered.
      68	 */
      69	function readToolResultText(content: string | unknown[]): string {
      70	  if (typeof content === "string") return content;
      71	
      72	  if (!Array.isArray(content)) {
      73	    throw new Error(
      74	      `tool result payload has no content to mark: expected a string or a content-block array, got ${typeof content}`,
      75	    );
      76	  }
      77	  if (content.length === 0) {
      78	    throw new Error(
      79	      "tool result payload has no content to mark: the content-block array is empty, so no payload was delivered",
      80	    );
      81	  }
      82	
      83	  const texts: string[] = [];
      84	  for (const rawBlock of content) {
      85	    const block = rawBlock as Record<string, unknown>;
      86	    if (block.type === "text") {
      87	      if (typeof block.text !== "string") {
      88	        throw new Error(
      89	          `tool result text block has no content to mark: expected a string, got ${typeof block.text}`,
      90	        );
      91	      }
      92	      texts.push(block.text);
      93	    } else if (block.type === "image") {
      94	      texts.push(IMAGE_UNAVAILABLE);
      95	    }
      96	  }
      97	
      98	  return texts.join("\n");
      99	}
