+/*
+STAS-399: inline-memory eviction must be observable by name, and its order must be declarable.
+
+Two defects, one fixture family:
+1. The clamp reported counts only ("44 of 62 older sections are NOT in this prompt"), so an agent
+   could not tell WHETHER a durable rule was among the lost, and the engine log line named nothing
+   either. Observability extends the notice the agent already reads; it is not a second report.
+2. The victim set was decided by file position: the OLDEST sections always went, whatever they held.
+   A `## ` section can now declare `sacrifice-first`; declared victims go before positional ones, so
+   a named rule sitting at the top of a file survives while expendable bulk goes instead.
+
+Every assertion is on the string returned by the real composition entry point
+(`resolveAgentInstructions`), never on source text, with the single deliberate exception of the
+one-definition case, which is a code-construct guard over the two files that could disagree.
+
+The length invariant is asserted PER CLAMPED SOURCE (one notice plus the body it introduces), not
+against the whole prompt: the clamp runs once per source (the inline `agent.memory` field and the
+workspace `MEMORY.md`), so a full-mode prompt legitimately carries two clamped sources plus identity,
+soul and instruction headers. That reading is the `ac2-length-invariant-scope` finding raised against
+this card on plan review; the whole-prompt length is printed, not asserted, for that reason.
+*/

+async function writeWorkspaceMemory(rootDir: string, content: string, agentId = "agent-test"): Promise<void> {
+  await mkdir(join(rootDir, ".fusion", "agent-memory", agentId), { recursive: true });
+  await writeFile(join(rootDir, ".fusion", "agent-memory", agentId, "MEMORY.md"), content, "utf-8");

+type Spec = { heading: string; body: number; marker?: string };
+
+/** One `## ` section whose whole body is `y` runs, so its rendered size is predictable to the char. */
+function section(spec: Spec): string {
+  return `## ${spec.heading}${spec.marker ?? ""}\n\n${"y".repeat(spec.body)}\n`;
+}
+
+function memoryDocument(specs: Spec[], preamble = "# Agent Memory\n"): string {
+  return [preamble, ...specs.map(section)].join("\n");
+}
+
+function withMarkers(specs: Spec[], marker: (heading: string) => string | undefined): Spec[] {
+  return specs.map((spec) => {
+    const declared = marker(spec.heading);
+    return declared ? { ...spec, marker: declared } : spec;
+  });
+}
+
+/**
+ * Sections whose rendered body is NOT in the prompt, derived from the render, never from the notice.
+ *
+ * The name returned is the heading text AFTER the `## ` prefix, which is what the shared parser
+ * (`parseMemoryFile`) yields and therefore what the notice is required to print — marker text
+ * included, because a `sacrifice-first` marker lives on the heading line and is part of that text.
+ * Returning the bare `spec.heading` would compare two different renderings of one heading and make
+ * the name-equality assertions unfalsifiable in one direction.
+ */
+function absentHeadings(rendered: string, specs: Spec[]): string[] {
+  return specs
+    .filter((spec) => !rendered.includes(`## ${spec.heading}${spec.marker ?? ""}\n\n${"y".repeat(spec.body)}`))
+    .map((spec) => `${spec.heading}${spec.marker ?? ""}`);
+}
+
+/**
+ * The clamp notice as one line, with the new trailing clause captured for parsing. Matched line-wise
+ * because a real heading may itself contain an underscore, the italic delimiter of the whole notice.
+ */
+const NOTICE =
+  /^_(Inline agent memory|Workspace memory): kept the newest (\d+) of (\d+) chars within the (\d+)-char inline memory budget, so (\d+) of (\d+) older sections \((\d+) chars\) are NOT in this prompt(.*)$/;
+
+function noticeLine(rendered: string, label: string): string {
+  return rendered.split("\n").find((line) => line.startsWith(`_${label}: kept the newest`)) ?? "";
+}
+
+function noticeFor(rendered: string, label: string): RegExpMatchArray | null {
+  return noticeLine(rendered, label).match(NOTICE) ?? null;
+}
+
+/** The body one notice introduces: everything after that notice line, up to the next notice or EOF. */
+function bodyAfterNotice(rendered: string, label: string): string {
+  const line = noticeLine(rendered, label);
+  if (!line) return "";
+  const start = rendered.indexOf(line) + line.length + 2;
+  const rest = rendered.slice(start);
+  const next = rest.search(/^_(?:Inline agent memory|Workspace memory): kept the newest/m);
+  return (next < 0 ? rest : next === 0 ? "" : rest.slice(0, next)).trimEnd();
+}
+
+function namesIn(tail: string): string[] {
+  const list = /: \[([^\]]*)\]/.exec(tail);
+  return list ? list[1].split(" | ").map((name) => name.trim()).filter(Boolean) : [];
+}
+
+function elidedNames(tail: string): number {
+  return Number(/\(\+(\d+) more names not listed\)/.exec(tail)?.[1] ?? 0);
+}
+
+function ignoredMarkers(tail: string): string[] {
+  const list = /sacrifice-first marker that is not a heading-line token: \[([^\]]*)\]/.exec(tail);
+  return list ? list[1].split(" | ").map((name) => name.trim()).filter(Boolean) : [];
+}
